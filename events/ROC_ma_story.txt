    id = ROC_ma_story.101
    title = ROC_ma_story.101.t
    desc = ROC_ma_story.101.d
    picture = GFX_report_event_generic_parliament
    is_triggered_only = yes
    timeout_days = 14
    option = {
        name = ROC_ma_story.101.a
        trigger = {
            ROC_ma_story_is_valid = yes
            check_variable = { var = ROC_ma_story_active_id value = 101 compare = equals }
        }
        ai_chance = { factor = 1 }
        custom_effect_tooltip = ROC_ma_story.101.a_tt
        hidden_effect = {
            if = {
                limit = {
                    ROC_ma_story_is_valid = yes
                    check_variable = { var = ROC_ma_story_active_id value = 101 compare = equals }
                }
                ROC_ma_story_clear = yes

                add_to_variable = { var = ROC_ce_pressure value = 3 }
                set_variable = { ROC_ma_day_delay = 14 }

        name = ROC_ma_story.101.b

        ai_chance = { factor = 1 }
        custom_effect_tooltip = ROC_ma_story.101.b_tt
        hidden_effect = {
            if = {
                limit = {
                    ROC_ma_story_is_valid = yes
                    check_variable = { var = ROC_ma_story_active_id value = 101 compare = equals }
                    NOT = { has_political_power < 10 }
                }
                ROC_ma_story_clear = yes
                add_political_power = -10
                add_to_variable = { var = ROC_ma_left_favor value = 6 }
                subtract_from_variable = { var = ROC_ma_right_favor value = 4 }
                set_variable = { ROC_ma_day_credit = 14 }
                ROC_ma_advance_days = yes
                ROC_ce_refresh = yes
            }
        }
    }
    option = {
        name = ROC_ma_story.101.c
        trigger = {
            ROC_ma_story_is_valid = yes
            check_variable = { var = ROC_ma_story_active_id value = 101 compare = equals }
            check_variable = { var = ROC_ma_center_favor value = 5 compare = greater_than_or_equals }
        }
        ai_chance = { factor = 1 }
        custom_effect_tooltip = ROC_ma_story.101.c_tt
        hidden_effect = {
            if = {
                limit = {
                    ROC_ma_story_is_valid = yes
                    check_variable = { var = ROC_ma_story_active_id value = 101 compare = equals }
                    check_variable = { var = ROC_ma_center_favor value = 5 compare = greater_than_or_equals }
                }
                ROC_ma_story_clear = yes
                subtract_from_variable = { var = ROC_ma_center_favor value = 5 }
                subtract_from_variable = { var = ROC_ma_left_favor value = 4 }
                subtract_from_variable = { var = ROC_ma_right_favor value = 3 }
                add_to_variable = { var = ROC_ce_actual value = 3 }
                ROC_ce_refresh = yes
            }
        }
    }
    option = {
        name = ROC_ma_story_archive
        trigger = {
            NOT = {

                }
            }
        }
        ai_chance = { factor = 100 }
        custom_effect_tooltip = ROC_ma_story_archive_tt
    }
}
