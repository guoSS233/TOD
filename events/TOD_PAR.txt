# 泛道山协约成立通知
country_event = {
	id = TOD_PAR.100
	title = TOD_PAR.100.t
	desc = TOD_PAR.100.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = PAR
		has_global_flag = TOD_nanmei_pandoushan_pact_signed
	}

	option = {
		name = TOD_PAR.100.a
		set_country_flag = TOD_PAR_pandoushan_pact_joined
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_in_faction_with = CHL }
			}
			leave_faction = yes
		}
		if = {
			limit = {
				country_exists = CHL
				CHL = { is_in_faction = yes }
				NOT = { is_in_faction_with = CHL }
			}
			CHL = { add_to_faction = PAR }
		}
	}
}
