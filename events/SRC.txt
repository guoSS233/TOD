#大会闭幕
country_event = {
	id = SRCevents.8
	title = SRCevents.8.t
	desc =SRCevents.8.d
	picture = GFX_report_event_chinese_soldiers_fighting
	
	fire_only_once = yes
	is_triggered_only = yes
	
	trigger = {tag = SRC}
	
	option = { 
		name = SRCevents.8.a
		trigger = {
			check_variable = { SRC_xianfeng > SRC_wenhe }
			check_variable = { SRC_xianfeng > SRC_weiyuan }
		}
		create_country_leader = SRC_chen_duxiu
		mark_focus_tree_layout_dirty = yes
	}
	option = { 
		name = SRCevents.8.b
		trigger = {
			check_variable = { SRC_weiyuan > SRC_xianfeng }
			check_variable = { SRC_weiyuan > SRC_wenhe }
		}
		create_country_leader = SRC_ren_fuchen
		mark_focus_tree_layout_dirty = yes
	}
	option = { 
		trigger = {
			check_variable = { SRC_xianfeng < SRC_wenhe }
			check_variable = { SRC_weiyuan < SRC_wenhe }
		}
		name = SRCevents.8.c
		create_country_leader = SRC_li_dazhao
		mark_focus_tree_layout_dirty = yes
	}
}
