# 1936 年春夏：整合西部及西伯利亚总督区，高加索、草原和远东留待后续阶段。
country_event = {
	id = TODRussia.2
	title = TODRussia.2.t
	desc = TODRussia.2.d
	picture = GFX_report_event_german_parade_paris
	fire_only_once = yes
	trigger = {
		tag = RUS
		date > 1936.2.28
		has_country_flag = TOD_RUS_election_finished
	}
	option = {
		name = TODRussia.2.a
		if = {
			limit = { country_exists = BLT }
			annex_country = { target = BLT transfer_troops = yes }
		}
		if = {
			limit = { country_exists = SIB }
			annex_country = { target = SIB transfer_troops = yes }
		}
		set_country_flag = TOD_RUS_western_governorates_integrated
	}
}

# 1938 年 3 月：高加索革命战争和“高加索红色阵线”。
country_event = {
	id = TODRussia.4
	title = TODRussia.4.t
	desc = TODRussia.4.d
	picture = GFX_report_event_chinese_soldiers_fighting
	fire_only_once = yes
	trigger = {
		tag = RUS
		date > 1938.2.28
		has_country_flag = TOD_RUS_asian_governorates_integrated
	}
	option = {
		name = TODRussia.4.a
		if = {
			limit = { country_exists = ARM }
			ARM = {
				if = {
					limit = { is_in_faction = no }
					create_faction = CAU_red_front
				}
			}
			if = {
				limit = { country_exists = GEO }
				GEO = { add_to_faction = ARM }
			}
			if = {
				limit = { country_exists = AZR }
				AZR = { add_to_faction = ARM }
			}
		}
		if = {
			limit = { country_exists = CAU }
			CAU = {
				if = {
					limit = { country_exists = ARM NOT = { has_war_with = ARM } }
					declare_war_on = { target = ARM type = annex_everything }
				}
				if = {
					limit = { country_exists = GEO NOT = { has_war_with = GEO } }
					declare_war_on = { target = GEO type = annex_everything }
				}
				if = {
					limit = { country_exists = AZR NOT = { has_war_with = AZR } }
					declare_war_on = { target = AZR type = annex_everything }
				}
			}
		}
		set_country_flag = TOD_RUS_caucasus_war_started
	}
}

	}
}
