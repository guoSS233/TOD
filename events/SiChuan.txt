					num_of_civilian_factories > 2
				}

					type = industrial_complex

				}
			}

				limit = { is_core_of = ROOT }
				add_building_construction = {
					type = industrial_complex
					level = 2
					instant_build = yes
				}
			}
		}
	}
	option = {
		name = TODSIC.1002.b
	}
