	id = WJP_politics.1
	title = WJP_politics.1.t
	desc = WJP_politics.1.d
	picture = GFX_report_event_generic_parliament
	fire_only_once = yes
	is_triggered_only = yes
	trigger = { tag = WJP }
	option = {
		name = WJP_politics.1.a
		custom_effect_tooltip = WJP_politics.1.effect
		hidden_effect = {
			if = {
				limit = { NOT = { has_country_flag = WJP_politics_model_regime_applied } }

				add_stability = 0.05
				remove_ideas = WJP_slave_of_dragon
				add_ideas = WJP_model_regime
			}
		}
	}
}
