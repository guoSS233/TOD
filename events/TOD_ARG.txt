# 胡斯托总统遇刺

# 拉普拉塔危机
country_event = {
	id = TOD_ARG.2
	title = TOD_ARG.2.t
	desc = TOD_ARG.2.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = ARG
		date > 1938.6.11
		has_country_flag = TOD_ARG_justo_assassinated
		NOT = { has_country_flag = TOD_ARG_la_plata_crisis_started }
	}

	option = {
		name = TOD_ARG.2.a
		set_country_flag = TOD_ARG_la_plata_crisis_started
		remove_ideas = ARG_extremist_currents
		remove_ideas = ARG_sharpen_bayonets
		add_ideas = ARG_extremist_politics
		add_ideas = ARG_drawn_bayonets
		set_politics = { ruling_party = national_corporatism elections_allowed = no }
		set_popularities = {
			national_corporatism = 70
			leader_despotism = 20
			authoritarian_oligarchy = 10
		}
		retire_country_leader = yes
		promote_character = ARG_Juan_Domingo_Peron
		add_war_support = 0.10
		add_stability = -0.05
		510 = { set_demilitarized_zone = no }
		511 = { set_demilitarized_zone = no }
		if = {
			limit = { country_exists = CHL NOT = { has_war_with = CHL } }
			declare_war_on = { target = CHL type = annex_everything }
		}
		if = {
			limit = { country_exists = PAR NOT = { has_war_with = PAR } }
			declare_war_on = { target = PAR type = annex_everything }
		}
		if = {
			limit = { country_exists = UPB NOT = { has_war_with = UPB } }
			declare_war_on = { target = UPB type = annex_everything }
		}
	}
}
