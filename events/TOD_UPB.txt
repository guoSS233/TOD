# 国王与内阁发生重大分歧
country_event = {
	id = TOD_UPB.1
	title = TOD_UPB.1.t
	desc = TOD_UPB.1.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = UPB
		date > 1937.5.1
		NOT = { has_country_flag = TOD_UPB_king_cabinet_conflict_started }
	}

	option = {
		name = TOD_UPB.1.a
		set_country_flag = TOD_UPB_king_cabinet_conflict_started
		remove_ideas = UPB_backward_reform
		add_ideas = UPB_emerging_reform
		add_political_power = -25
		add_stability = -0.03
		country_event = { id = TOD_UPB.2 days = 40 }
	}
}

# 泛道山协约成立通知
country_event = {
	id = TOD_UPB.100
	title = TOD_UPB.100.t
	desc = TOD_UPB.100.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = UPB
		has_global_flag = TOD_nanmei_pandoushan_pact_signed
	}

	option = {
		name = TOD_UPB.100.a
		set_country_flag = TOD_UPB_pandoushan_pact_joined
		if = {
			limit = {
				is_in_faction = yes
				NOT = { is_in_faction_with = CHL }
			}
			leave_faction = yes
		}
		if = {
			limit = {
				country_exists = CHL
				CHL = { is_in_faction = yes }
				NOT = { is_in_faction_with = CHL }
			}
			CHL = { add_to_faction = UPB }
		}
	}
}
