#决战沈阳
country_event = {
	id = GPSevents.9
	title = GPSevents.9.t
	desc =GPSevents.9.d
	picture = GFX_report_event_chinese_soldiers_fighting
	
	fire_only_once = yes
	is_triggered_only = yes
	
	trigger = {tag = GPS}
	
	option = { 
		name = GPSevents.9.a
        if = {
			limit = {check_variable = { GPS_sheng > GPS_bai}}
            GPS = {country_event = { id = GPSevents.10 days = 2 } }
		}
        if = {
			limit = {check_variable = { GPS_sheng = GPS_bai}}
            GPS = {country_event = { id = GPSevents.10 days = 2 } }
		}
        if = {
			limit = {check_variable = { GPS_sheng < GPS_bai}}
            GPS = {country_event = { id = GPSevents.11 days = 2 } }
		}
	}
	option = { 
		name = GPSevents.9.b
        if = {
			limit = {check_variable = { GPS_sheng > GPS_bai}}
            GPS = {country_event = { id = GPSevents.10 days = 2 } }
		}
        if = {
			limit = {check_variable = { GPS_sheng = GPS_bai}}
            GPS = {country_event = { id = GPSevents.11 days = 2 } }
		}
        if = {
			limit = {check_variable = { GPS_sheng < GPS_bai}}
            GPS = {country_event = { id = GPSevents.11 days = 2 } }
		}
	}
}
