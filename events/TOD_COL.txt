		NOT = { has_country_flag = TOD_COL_popayan_accord_signed }
	}

	option = {
		name = TOD_COL.4.a
		set_country_flag = TOD_COL_popayan_accord_signed
		set_global_flag = TOD_nanmei_revenge_alliance_created
		set_rule = { can_create_factions = yes }
		if = {
			limit = { NOT = { is_in_faction = yes } }

			limit = { country_exists = BRA }

				set_country_flag = TOD_BRA_popayan_accord_signed
				country_event = { id = TOD_BRA.2003 days = 2 }
			}
		}
	}

# 哥伦比亚-巴西对南华自由邦宣战

	id = TOD_COL.5
	title = TOD_COL.5.t
	desc = TOD_COL.5.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = COL
		date > 1940.2.13
		has_country_flag = TOD_COL_popayan_accord_signed
		country_exists = GYA
		NOT = { has_country_flag = TOD_COL_war_against_gya_started }
	}

	option = {
		name = TOD_COL.5.a
		set_country_flag = TOD_COL_war_against_gya_started
		set_global_flag = TOD_nanmei_revenge_alliance_war_started
		declare_war_on = { target = GYA type = annex_everything }
		if = {
			limit = {
				country_exists = BRA
				BRA = { NOT = { has_war_with = GYA } }
			}
			BRA = {
				set_country_flag = TOD_BRA_war_against_gya_started
				country_event = { id = TOD_BRA.2004 days = 2 }
				declare_war_on = { target = GYA type = annex_everything }
			}
		}
	}
}
