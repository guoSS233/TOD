add_namespace = SRC_debug

country_event = {
    id = SRC_debug.1
    title = SRC_debug.1.t
    desc = SRC_debug.1.d
    is_triggered_only = yes

    option = {
        name = SRC_debug.1.a
        if = {
            limit = { country_exists = SRC }
            SRC = {
                complete_national_focus = SRC_lianluoguanneigemingzhe
                add_political_power = 300
                clr_country_flag = SRC_huabeizhongyangju_established
                clr_country_flag = SRC_huabeizhongyangju_active
                set_country_flag = SRC_zhongyangju_gui_open
            }
        }
    }

    option = {
        name = SRC_debug.1.b
        if = {
            limit = { country_exists = SRC }
            SRC = {
                complete_national_focus = SRC_lianluoguanneigemingzhe
                add_political_power = 300
                set_country_flag = SRC_huabeizhongyangju_established
                set_country_flag = SRC_huabeizhongyangju_active
                set_country_flag = SRC_zhongyangju_gui_open
                if = {
                    limit = { has_idea = SRC_huabeizhongyangju_idea }
                    remove_ideas = SRC_huabeizhongyangju_idea
                }
                add_ideas = SRC_huabeizhongyangju_idea
            }
        }
    }

    option = {
        name = SRC_debug.1.c
    }
}
