# 泛道山协约成立
country_event = {
	id = TOD_CHL.100
	title = TOD_CHL.100.t
	desc = TOD_CHL.100.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = CHL
		date > 1938.6.11
		country_exists = UPB
		country_exists = PAR
		NOT = { has_country_flag = TOD_CHL_pandoushan_pact_signed }
	}

	option = {
		name = TOD_CHL.100.a
		set_country_flag = TOD_CHL_pandoushan_pact_signed
		set_global_flag = TOD_nanmei_pandoushan_pact_signed
		if = {
			limit = { is_in_faction = yes }
			leave_faction = yes
		}
		create_faction = pandoushan_pact
		if = {
			limit = { country_exists = UPB }
			UPB = { country_event = { id = TOD_UPB.100 days = 1 } }
		}
		if = {
			limit = { country_exists = PAR }
			PAR = { country_event = { id = TOD_PAR.100 days = 1 } }
		}
	}
}
# 泛道山协约对复仇同盟宣战
country_event = {
	id = TOD_CHL.101
	title = TOD_CHL.101.t
	desc = TOD_CHL.101.d
	picture = GFX_report_event_chinese_soldiers_fighting

	fire_only_once = yes
	is_triggered_only = yes

	trigger = {
		tag = CHL
		date > 1940.3.4
		has_country_flag = TOD_CHL_pandoushan_pact_signed
		has_global_flag = TOD_nanmei_pandoushan_revenge_war_signal
		OR = {
			AND = { country_exists = BRA NOT = { has_war_with = BRA } }
			AND = { country_exists = COL NOT = { has_war_with = COL } }
		}
		NOT = { has_country_flag = TOD_CHL_pandoushan_revenge_war_started }
	}

	option = {
		name = TOD_CHL.101.a
		set_country_flag = TOD_CHL_pandoushan_revenge_war_started
		set_global_flag = TOD_nanmei_pandoushan_revenge_war_started
		add_war_support = 0.10
		add_stability = -0.03
		if = {
			limit = { country_exists = BRA NOT = { has_war_with = BRA } }
			declare_war_on = { target = BRA type = annex_everything }
		}
		if = {
			limit = { country_exists = COL NOT = { has_war_with = COL } }
			declare_war_on = { target = COL type = annex_everything }
		}
		if = {
			limit = { country_exists = UPB }
			UPB = {
				set_country_flag = TOD_UPB_pandoushan_revenge_war_started
				add_war_support = 0.05
				if = {
					limit = { is_in_faction_with = CHL country_exists = BRA NOT = { has_war_with = BRA } }
					declare_war_on = { target = BRA type = annex_everything }
				}
				if = {
					limit = { is_in_faction_with = CHL country_exists = COL NOT = { has_war_with = COL } }
					declare_war_on = { target = COL type = annex_everything }
				}
			}
		}
		if = {
			limit = { country_exists = PAR }
			PAR = {
				set_country_flag = TOD_PAR_pandoushan_revenge_war_started
				add_war_support = 0.05
				if = {
					limit = { is_in_faction_with = CHL country_exists = BRA NOT = { has_war_with = BRA } }
					declare_war_on = { target = BRA type = annex_everything }
				}
				if = {
					limit = { is_in_faction_with = CHL country_exists = COL NOT = { has_war_with = COL } }
					declare_war_on = { target = COL type = annex_everything }
				}
			}
		}
	}
}
