ROC_committee_daily = {

    clamp_variable = { var = ROC_committee_radicalization min = 0 max = 100 }
    if = {
        limit = { NOT = { has_country_flag = ROC_committee_initialized } }
        # Reading an old save restores completed counts, without charging another weekly reaction.
        set_country_flag = ROC_committee_initialized
        set_country_flag = { flag = ROC_committee_weekly_tick days = 7 }
    }
    if = {
        limit = { ROC_econ_race_active = yes }
        ROC_committee_discard_completed_agendas = yes
        ROC_committee_select_left = yes
        ROC_committee_select_center = yes
        ROC_committee_select_right = yes
        # Adopt an existing countdown without reactivating it or changing its duration.
        if = { limit = { NOT = { has_active_mission = ROC_decision_zuopai } check_variable = { ROC_committee_left_agenda > 0 } } activate_mission = ROC_decision_zuopai }

    else = { ROC_econ_finish_race = yes }
    ROC_econ_refresh = yes
