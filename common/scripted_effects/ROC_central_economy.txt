ROC_ce_initialize = {
    # Read the legacy initialized flag before first adoption sets it.
    ROC_ce_sync_minister_profile = yes
    if = {
        limit = { ROC_ce_administration_active = yes NOT = { has_country_flag = ROC_ce_initialized } }
        set_country_flag = ROC_ce_initialized
        # A legacy save starts its new performance account today, without invented output.
        set_variable = { ROC_ce_actual = 0 }
        set_variable = { ROC_ce_expected = 0 }

        set_variable = { ROC_ce_reviews_behind = 0 }

}

    ROC_ce_policy_initialize = yes
    if = {
        limit = { has_power_balance = { id = ROC_power_balance } }
        # Removal may execute old range callbacks. Restore exact existing political clocks.
        if = { limit = { has_active_mission = ROC_decision_zuopai } set_variable = { ROC_ce_saved_left = days_mission_timeout@ROC_decision_zuopai } }
        if = { limit = { has_active_mission = ROC_decision_zhongpai } set_variable = { ROC_ce_saved_center = days_mission_timeout@ROC_decision_zhongpai } }

    clr_country_flag = ROC_flag_bop_jihua_3

ROC_ce_deliver = {
    if = {
        limit = { ROC_ce_current_project = yes check_variable = { var = ROC_ce_project_progress value = 100 compare = greater_than_or_equals } }

    clr_country_flag = ROC_ce_delivery_valid
    if = {
        limit = { check_variable = { ROC_ce_project_id = 1 } }
        random_owned_controlled_state = {
            limit = { ROC_ce_industry_target = yes }
            add_building_construction = { type = industrial_complex level = 1 instant_build = yes }
            ROOT = { set_country_flag = ROC_ce_delivery_valid }
        }
    }
    else_if = {
        limit = { OR = { check_variable = { ROC_ce_project_id = 2 } check_variable = { ROC_ce_project_id = 3 } } }
        random_owned_controlled_state = {
            limit = { ROC_ce_infrastructure_target = yes }
            add_building_construction = { type = infrastructure level = 1 instant_build = yes }
            ROOT = { set_country_flag = ROC_ce_delivery_valid }
        }
    }
    if = {
        limit = { has_country_flag = ROC_ce_delivery_valid }
        if = { limit = { check_variable = { ROC_ce_project_id = 2 } } add_stability = 0.01 }
        add_to_variable = { var = ROC_ce_actual value = 20 }
        add_to_variable = { var = ROC_ce_projects_completed value = 1 }
        subtract_from_variable = { var = ROC_ce_pressure value = 8 }
        add_to_variable = { var = ROC_ma_center_favor value = 5 }
        set_variable = { ROC_ma_day_credit = 30 }
        ROC_ma_advance_days = yes
        set_variable = { ROC_ce_project_id = 0 }
        set_variable = { ROC_ce_project_progress = 100 }
        ROC_ce_clear_project_windows = yes
        clr_country_flag = ROC_ce_no_delivery_target
    }
    else = {
        set_variable = { ROC_ce_project_progress = 99 }
        set_country_flag = ROC_ce_no_delivery_target
    }
    ROC_ce_refresh = yes

    }
}
