# Hidden axes: foundation -100 mass tradition / +100 professional officers.
# Authority -100 soldier autonomy / +100 party leadership.
# Visible construction values are independent of both axes.
SRC_wartime_army_clamp = {
	clamp_variable = { var = SRC_wartime_army_command_stage min = 0 max = 100 }
	clamp_variable = { var = SRC_wartime_army_training_stage min = 0 max = 100 }
	clamp_variable = { var = SRC_wartime_army_political_stage min = 0 max = 100 }
	clamp_variable = { var = SRC_wartime_army_supply_stage min = 0 max = 100 }
	clamp_variable = { var = SRC_army_foundation min = -100 max = 100 }
	clamp_variable = { var = SRC_army_authority min = -100 max = 100 }
	add_to_variable = { SRC_wartime_army_gui_dirty = 0.01 }
}

SRC_wartime_army_recalculate = {
	set_variable = { SRC_wartime_army_average = SRC_wartime_army_command_stage }
	add_to_variable = { SRC_wartime_army_average = SRC_wartime_army_training_stage }
	add_to_variable = { SRC_wartime_army_average = SRC_wartime_army_political_stage }
	add_to_variable = { SRC_wartime_army_average = SRC_wartime_army_supply_stage }
	divide_variable = { SRC_wartime_army_average = 4 }

	set_variable = { SRC_wartime_army_values_at_60 = 0 }
	if = { limit = { OR = { check_variable = { SRC_wartime_army_command_stage > 60 } check_variable = { SRC_wartime_army_command_stage = 60 } } } add_to_variable = { SRC_wartime_army_values_at_60 = 1 } }
	if = { limit = { OR = { check_variable = { SRC_wartime_army_training_stage > 60 } check_variable = { SRC_wartime_army_training_stage = 60 } } } add_to_variable = { SRC_wartime_army_values_at_60 = 1 } }
	if = { limit = { OR = { check_variable = { SRC_wartime_army_political_stage > 60 } check_variable = { SRC_wartime_army_political_stage = 60 } } } add_to_variable = { SRC_wartime_army_values_at_60 = 1 } }
	if = { limit = { OR = { check_variable = { SRC_wartime_army_supply_stage > 60 } check_variable = { SRC_wartime_army_supply_stage = 60 } } } add_to_variable = { SRC_wartime_army_values_at_60 = 1 } }
}

SRC_wartime_army_refresh_spirit = {
	SRC_wartime_army_clamp = yes
	SRC_wartime_army_recalculate = yes
	if = {
		limit = { NOT = { SRC_wartime_army_has_final_spirit = yes } }
		if = {
			limit = { SRC_wartime_army_ready_for_northeast_red_army = yes }
			if = {
				limit = { NOT = { has_idea = SRC_northeast_red_army } }
				SRC_wartime_army_remove_stage_spirits = yes
				add_ideas = SRC_northeast_red_army
			}
		}
		else_if = {
			limit = { SRC_wartime_army_ready_for_formed_army = yes }
			if = {
				limit = { NOT = { has_idea = SRC_formed_northeast_red_army } }
				SRC_wartime_army_remove_stage_spirits = yes
				add_ideas = SRC_formed_northeast_red_army
			}
		}
		else_if = {
			limit = { SRC_wartime_army_ready_for_initial_integration = yes }
			if = {
				limit = { NOT = { has_idea = SRC_initially_integrated_red_army } }
				SRC_wartime_army_remove_stage_spirits = yes
				add_ideas = SRC_initially_integrated_red_army
			}
		}
		else = {
			if = {
				limit = { NOT = { has_idea = SRC_improvised_revolutionary_force } }
				SRC_wartime_army_remove_stage_spirits = yes
				add_ideas = SRC_improvised_revolutionary_force
			}
		}
	}
	add_to_variable = { SRC_wartime_army_gui_dirty = 0.01 }
}
