# Alliance parliament seats are persistent state. Initialization only seeds a new
# country; refresh derives percentage-point display values from the live totals.
WJP_parliament_initialize_effect = {
	if = {
		limit = { NOT = { has_country_flag = WJP_parliament_initialized } }
		set_variable = { WJP_parliament_free_society_seats = 124 }
		set_variable = { WJP_parliament_progressive_association_seats = 64 }
		set_variable = { WJP_parliament_national_renewal_league_seats = 23 }
		set_variable = { WJP_parliament_daido_association_seats = 16 }
		set_variable = { WJP_parliament_seat_frame = 1 }
		set_country_flag = WJP_parliament_initialized
	}
	WJP_parliament_refresh_effect = yes
}

WJP_parliament_refresh_effect = {
	set_variable = { var = WJP_parliament_free_society_percent value = WJP_parliament_free_society_seats }
	multiply_variable = { var = WJP_parliament_free_society_percent value = 100 }
	divide_variable = { var = WJP_parliament_free_society_percent value = 227 }
	set_variable = { var = WJP_parliament_progressive_association_percent value = WJP_parliament_progressive_association_seats }
	multiply_variable = { var = WJP_parliament_progressive_association_percent value = 100 }
	divide_variable = { var = WJP_parliament_progressive_association_percent value = 227 }
	set_variable = { var = WJP_parliament_national_renewal_league_percent value = WJP_parliament_national_renewal_league_seats }
	multiply_variable = { var = WJP_parliament_national_renewal_league_percent value = 100 }
	divide_variable = { var = WJP_parliament_national_renewal_league_percent value = 227 }
	set_variable = { var = WJP_parliament_daido_association_percent value = WJP_parliament_daido_association_seats }
	multiply_variable = { var = WJP_parliament_daido_association_percent value = 100 }
	divide_variable = { var = WJP_parliament_daido_association_percent value = 227 }
}

WJP_han_assembly_initialize_effect = {
	if = {
		limit = { NOT = { has_country_flag = WJP_han_assembly_initialized } }
		set_variable = { WJP_han_tosa_choshu = 0 }
		set_variable = { WJP_han_tosa_satsuma = 0 }
		set_variable = { WJP_han_hizen_choshu = 1 }
		set_variable = { WJP_han_hizen_satsuma = 0 }
		set_variable = { WJP_han_buzen_choshu = 1 }
		set_variable = { WJP_han_buzen_satsuma = 0 }
		set_variable = { WJP_han_izumo_choshu = 0 }
		set_variable = { WJP_han_izumo_satsuma = 1 }
		set_variable = { WJP_han_harima_choshu = 0 }
		set_variable = { WJP_han_harima_satsuma = 0 }
		add_ideas = WJP_han_inclination_choshu
		set_country_flag = WJP_han_assembly_initialized
	}
	WJP_han_assembly_refresh_effect = yes
	WJP_han_assembly_maintain_mission_effect = yes
}

WJP_han_assembly_refresh_effect = {
	set_variable = { WJP_han_choshu_domain_count = 1 }
	set_variable = { WJP_han_satsuma_domain_count = 1 }
	if = {
		limit = { check_variable = { WJP_han_tosa_choshu > WJP_han_tosa_satsuma } }
		add_to_variable = { WJP_han_choshu_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_hizen_choshu > WJP_han_hizen_satsuma } }
		add_to_variable = { WJP_han_choshu_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_buzen_choshu > WJP_han_buzen_satsuma } }
		add_to_variable = { WJP_han_choshu_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_izumo_choshu > WJP_han_izumo_satsuma } }
		add_to_variable = { WJP_han_choshu_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_harima_choshu > WJP_han_harima_satsuma } }
		add_to_variable = { WJP_han_choshu_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_tosa_satsuma > WJP_han_tosa_choshu } }
		add_to_variable = { WJP_han_satsuma_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_hizen_satsuma > WJP_han_hizen_choshu } }
		add_to_variable = { WJP_han_satsuma_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_buzen_satsuma > WJP_han_buzen_choshu } }
		add_to_variable = { WJP_han_satsuma_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_izumo_satsuma > WJP_han_izumo_choshu } }
		add_to_variable = { WJP_han_satsuma_domain_count = 1 }
	}
	if = {
		limit = { check_variable = { WJP_han_harima_satsuma > WJP_han_harima_choshu } }
		add_to_variable = { WJP_han_satsuma_domain_count = 1 }
	}
	set_variable = { WJP_han_neutral_domain_count = 7 }
	subtract_from_variable = { WJP_han_neutral_domain_count = WJP_han_choshu_domain_count }
	subtract_from_variable = { WJP_han_neutral_domain_count = WJP_han_satsuma_domain_count }
}

WJP_han_assembly_resolve_effect = {
	WJP_han_assembly_refresh_effect = yes
	if = {
		limit = { check_variable = { WJP_han_choshu_domain_count > WJP_han_satsuma_domain_count } }
		remove_ideas = WJP_han_inclination_satsuma
		add_ideas = WJP_han_inclination_choshu
	}
	else_if = {
		limit = { check_variable = { WJP_han_satsuma_domain_count > WJP_han_choshu_domain_count } }
		remove_ideas = WJP_han_inclination_choshu
		add_ideas = WJP_han_inclination_satsuma
	}
	WJP_han_assembly_refresh_effect = yes
}


# Repeated timeout activation is explicit; daily/startup repair never resets an active timer.
WJP_han_assembly_maintain_mission_effect = {
	if = {
		limit = {
			has_country_flag = WJP_han_assembly_initialized
			NOT = { has_active_mission = WJP_han_assembly_leadership_resolution }
		}
		activate_mission = WJP_han_assembly_leadership_resolution
	}
}

# The archive contains only the one budget bill already handled by the review.
# Live passage records the in-game year before any later maintenance can run.
WJP_parliament_budget_archive_passed_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { var = WJP_parliament_budget_archive_year value = global.year }
		WJP_parliament_budget_archive_snapshot_effect = yes
	}
}

# Older saves may already have a passed result but no archive. The original year
# is unknowable there, so zero is kept as an explicit unknown value.
WJP_parliament_budget_archive_migrate_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { WJP_parliament_budget_archive_year = 0 }
		WJP_parliament_budget_archive_snapshot_effect = yes
	}
}

WJP_parliament_budget_archive_snapshot_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { WJP_parliament_budget_archive_plan = 0 }
		set_variable = { WJP_parliament_budget_archive_budget = 0 }
		if = {
			limit = { has_country_flag = WJP_economic_plan_minsei }
			set_variable = { WJP_parliament_budget_archive_plan = 1 }
		}
		else_if = {
			limit = { has_country_flag = WJP_economic_plan_kokumu }
			set_variable = { WJP_parliament_budget_archive_plan = 2 }
		}
		if = {
			limit = { has_country_flag = WJP_parliament_budget_minsei }
			set_variable = { WJP_parliament_budget_archive_budget = 1 }
		}
		else_if = {
			limit = { has_country_flag = WJP_parliament_budget_kokumu }
			set_variable = { WJP_parliament_budget_archive_budget = 2 }
		}
		set_country_flag = WJP_parliament_budget_archived
	}
}

# Party attitudes persist across reloads and remain bounded to five levels.
WJP_parliament_attitudes_initialize_effect = {
	if = {
		limit = { NOT = { has_country_flag = WJP_parliament_attitudes_initialized } }
		set_variable = { WJP_parliament_free_society_attitude = 2 }
		set_variable = { WJP_parliament_progressive_association_attitude = -2 }
		set_variable = { WJP_parliament_national_renewal_league_attitude = -1 }
		set_variable = { WJP_parliament_daido_association_attitude = 0 }
		set_country_flag = WJP_parliament_attitudes_initialized
	}
}

WJP_parliament_attitudes_refresh_effect = {
	if = {
		limit = { check_variable = { WJP_parliament_free_society_attitude < -2 } }
		set_variable = { WJP_parliament_free_society_attitude = -2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_free_society_attitude > 2 } }
		set_variable = { WJP_parliament_free_society_attitude = 2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_progressive_association_attitude < -2 } }
		set_variable = { WJP_parliament_progressive_association_attitude = -2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_progressive_association_attitude > 2 } }
		set_variable = { WJP_parliament_progressive_association_attitude = 2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_national_renewal_league_attitude < -2 } }
		set_variable = { WJP_parliament_national_renewal_league_attitude = -2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_national_renewal_league_attitude > 2 } }
		set_variable = { WJP_parliament_national_renewal_league_attitude = 2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_daido_association_attitude < -2 } }
		set_variable = { WJP_parliament_daido_association_attitude = -2 }
	}
	if = {
		limit = { check_variable = { WJP_parliament_daido_association_attitude > 2 } }
		set_variable = { WJP_parliament_daido_association_attitude = 2 }
	}

	set_variable = { WJP_parliament_supporting_parties = 0 }
	set_variable = { WJP_parliament_extreme_opponents = 0 }
	if = {
		limit = { check_variable = { var = WJP_parliament_free_society_attitude value = 1 compare = greater_than_or_equals } }
		add_to_variable = { WJP_parliament_supporting_parties = 1 }
	}
	if = {
		limit = { check_variable = { var = WJP_parliament_free_society_attitude value = -2 compare = less_than_or_equals } }
		add_to_variable = { WJP_parliament_extreme_opponents = 1 }
	}
	if = {
		limit = { check_variable = { var = WJP_parliament_progressive_association_attitude value = 1 compare = greater_than_or_equals } }
		add_to_variable = { WJP_parliament_supporting_parties = 1 }
	}
	if = {
		limit = { check_variable = { var = WJP_parliament_progressive_association_attitude value = -2 compare = less_than_or_equals } }
		add_to_variable = { WJP_parliament_extreme_opponents = 1 }
	}
	if = {
		limit = { check_variable = { var = WJP_parliament_national_renewal_league_attitude value = 1 compare = greater_than_or_equals } }

# One native mission owns the deadline; maintenance only reads its live timer.
WJP_parliament_begin_deliberation_effect = {
	if = {
		limit = {
			has_country_flag = WJP_economic_bill_submitted
			NOT = { has_country_flag = WJP_parliament_bill_resolved }
			NOT = { has_country_flag = WJP_parliament_deliberation_started }
		}
		WJP_parliament_attitudes_initialize_effect = yes
		WJP_parliament_attitudes_refresh_effect = yes
		set_country_flag = WJP_parliament_deliberation_started
		set_country_flag = WJP_parliament_bill_active
	}
	if = {
		limit = {
			has_country_flag = WJP_economic_bill_submitted
			NOT = { has_country_flag = WJP_parliament_bill_resolved }
			has_country_flag = WJP_parliament_deliberation_started
			has_country_flag = WJP_parliament_bill_active
		}
		if = {
			limit = { NOT = { has_active_mission = WJP_parliament_bill_deliberation } }
			activate_mission = WJP_parliament_bill_deliberation
		}
		if = {
			limit = { NOT = { has_country_flag = WJP_parliament_deliberation_events_queued } }
			set_country_flag = WJP_parliament_deliberation_events_queued
			country_event = { id = WJP_parliament.1 days = 12 }
			country_event = { id = WJP_parliament.2 days = 30 }
		}
		if = {
			limit = { has_active_mission = WJP_parliament_bill_deliberation }
			set_variable = { WJP_parliament_review_days = days_mission_timeout@WJP_parliament_bill_deliberation }
		}
	}
}

WJP_parliament_bill_maintenance_effect = {
	WJP_parliament_attitudes_initialize_effect = yes
	WJP_parliament_attitudes_refresh_effect = yes
	WJP_parliament_budget_archive_migrate_effect = yes
	if = {
		limit = {
			has_country_flag = WJP_economic_bill_submitted
			NOT = { has_country_flag = WJP_parliament_bill_resolved }
		}
		WJP_parliament_begin_deliberation_effect = yes
	}
	if = {
		limit = {
			has_active_mission = WJP_parliament_bill_deliberation
			NOT = { has_country_flag = WJP_parliament_bill_resolved }
		}
		set_variable = { WJP_parliament_review_days = days_mission_timeout@WJP_parliament_bill_deliberation }
	}
	else = {
		set_variable = { WJP_parliament_review_days = 0 }
	}
}

WJP_parliament_bill_resolve_timeout_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_active
			NOT = { has_country_flag = WJP_parliament_bill_resolved }
		}
		WJP_parliament_attitudes_refresh_effect = yes
		if = {
			limit = { WJP_parliament_bill_can_pass = yes }
			set_country_flag = WJP_parliament_bill_resolved
			set_country_flag = WJP_parliament_bill_passed
			WJP_parliament_budget_archive_passed_effect = yes
			clr_country_flag = WJP_parliament_bill_active
			clr_country_flag = WJP_parliament_bill_rejected
			set_variable = { WJP_parliament_review_days = 0 }
			country_event = { id = WJP_parliament.100 days = 1 }
		}
		else = {
			set_country_flag = WJP_parliament_bill_resolved
			set_country_flag = WJP_parliament_bill_rejected
			clr_country_flag = WJP_parliament_bill_active
			clr_country_flag = WJP_parliament_bill_passed
			set_variable = { WJP_parliament_review_days = 0 }
			country_event = { id = WJP_parliament.101 days = 1 }
		}
	}
}
