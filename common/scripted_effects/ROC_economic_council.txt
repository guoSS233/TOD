ROC_econ_refresh = {
    set_variable = { ROC_econ_term_remaining = 0 }
    set_variable = { ROC_econ_term_progress = 0 }
    set_variable = { ROC_econ_stage = 0 }
    if = { limit = { has_active_mission = ROC_decision_bop_jingjidahui NOT = { has_country_flag = ROC_econ_conference_pending } } set_country_flag = ROC_econ_term_active set_variable = { ROC_econ_term_remaining = days_mission_timeout@ROC_decision_bop_jingjidahui } set_variable = { ROC_econ_term_progress = 1095 } subtract_from_variable = { var = ROC_econ_term_progress value = ROC_econ_term_remaining } divide_variable = { var = ROC_econ_term_progress value = 1095 } clamp_variable = { var = ROC_econ_term_progress min = 0 max = 1 } if = { limit = { check_variable = { ROC_econ_term_remaining > 730 } } set_variable = { ROC_econ_stage = 1 } } else_if = { limit = { check_variable = { ROC_econ_term_remaining > 365 } } set_variable = { ROC_econ_stage = 2 } } else = { set_variable = { ROC_econ_stage = 3 } } }
    set_variable = { ROC_econ_axis_percent = 0 }
    set_variable = { ROC_econ_axis_progress = 0.5 }

    ROC_econ_queue_shift = yes
    set_variable = { ROC_econ_pending_percent = ROC_econ_pending_shift }
    multiply_variable = { var = ROC_econ_pending_percent value = 100 }
    set_variable = { ROC_econ_left_days = 0 }
    set_variable = { ROC_econ_center_days = 0 }
    set_variable = { ROC_econ_right_days = 0 }
    if = { limit = { has_active_mission = ROC_decision_zuopai } set_variable = { ROC_econ_left_days = days_mission_timeout@ROC_decision_zuopai } }
    if = { limit = { has_active_mission = ROC_decision_zhongpai } set_variable = { ROC_econ_center_days = days_mission_timeout@ROC_decision_zhongpai } }
    if = { limit = { has_active_mission = ROC_decision_youpai } set_variable = { ROC_econ_right_days = days_mission_timeout@ROC_decision_youpai } }
    # Progressbar frames use 0..100; retain the 0..1 ratios for text.
    set_variable = { ROC_econ_term_frame = ROC_econ_term_progress }
    multiply_variable = { var = ROC_econ_term_frame value = 100 }
    clamp_variable = { var = ROC_econ_term_frame min = 0 max = 100 }
    set_variable = { ROC_econ_axis_frame = ROC_econ_axis_progress }
    multiply_variable = { var = ROC_econ_axis_frame value = 100 }
    clamp_variable = { var = ROC_econ_axis_frame min = 0 max = 100 }
}
