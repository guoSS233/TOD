# The archive contains only the one budget bill already handled by the review.
# Live passage records the in-game year before any later maintenance can run.
WJP_parliament_budget_archive_passed_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { var = WJP_parliament_budget_archive_year value = global.year }
		WJP_parliament_budget_archive_snapshot_effect = yes
	}
}

# Older saves may already have a passed result but no archive. The original year
# is unknowable there, so zero is kept as an explicit unknown value.
WJP_parliament_budget_archive_migrate_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { WJP_parliament_budget_archive_year = 0 }
		WJP_parliament_budget_archive_snapshot_effect = yes
	}
}

WJP_parliament_budget_archive_snapshot_effect = {
	if = {
		limit = {
			has_country_flag = WJP_parliament_bill_passed
			NOT = { has_country_flag = WJP_parliament_budget_archived }
		}
		set_variable = { WJP_parliament_budget_archive_plan = 0 }
		set_variable = { WJP_parliament_budget_archive_budget = 0 }
		if = {
			limit = { has_country_flag = WJP_economic_plan_minsei }
			set_variable = { WJP_parliament_budget_archive_plan = 1 }
		}
		else_if = {
			limit = { has_country_flag = WJP_economic_plan_kokumu }
			set_variable = { WJP_parliament_budget_archive_plan = 2 }
		}
		if = {
			limit = { has_country_flag = WJP_parliament_budget_minsei }
			set_variable = { WJP_parliament_budget_archive_budget = 1 }
		}
		else_if = {
			limit = { has_country_flag = WJP_parliament_budget_kokumu }
			set_variable = { WJP_parliament_budget_archive_budget = 2 }
		}
		set_country_flag = WJP_parliament_budget_archived
	}
}
