# Existing SRC military focuses only. Each role group owns one focus and one work counter.
# 140 work points per focus: one member adds 1/day; two matching army members add 2/day.
SRC_inner_circle_auto_focus_daily = {
    if = {
        limit = { NOT = { has_country_flag = SRC_inner_circle_auto_initialized } }
        set_variable = { SRC_inner_circle_auto_soldiers_target = 0 }
        set_variable = { SRC_inner_circle_auto_soldiers_work = 0 }
        set_variable = { SRC_inner_circle_auto_commissars_target = 0 }
        set_variable = { SRC_inner_circle_auto_commissars_work = 0 }
        set_variable = { SRC_inner_circle_auto_commanders_target = 0 }
        set_variable = { SRC_inner_circle_auto_commanders_work = 0 }
        set_variable = { SRC_inner_circle_auto_reform_target = 0 }
        set_variable = { SRC_inner_circle_auto_reform_work = 0 }
        set_variable = { SRC_inner_circle_auto_navy_target = 0 }
        set_variable = { SRC_inner_circle_auto_navy_work = 0 }
        set_variable = { SRC_inner_circle_auto_air_target = 0 }
        set_variable = { SRC_inner_circle_auto_air_work = 0 }
        set_variable = { SRC_inner_circle_auto_reform_branch = 0 }
        set_country_flag = SRC_inner_circle_auto_initialized
    }
    if = {
        limit = { NOT = { has_country_flag = SRC_inner_circle_auto_daily_fix_v1 } }
        # The old global on_daily hook advanced SRC once per country each day.
        # Reset unfinished work once so existing saves use real elapsed days.
        set_variable = { SRC_inner_circle_auto_soldiers_work = 0 }
        set_variable = { SRC_inner_circle_auto_commissars_work = 0 }
        set_variable = { SRC_inner_circle_auto_commanders_work = 0 }
        set_variable = { SRC_inner_circle_auto_reform_work = 0 }
        set_variable = { SRC_inner_circle_auto_navy_work = 0 }
        set_variable = { SRC_inner_circle_auto_air_work = 0 }
        set_country_flag = SRC_inner_circle_auto_daily_fix_v1
    }
    if = {
        limit = { NOT = { has_country_flag = SRC_inner_circle_auto_thought_focus_v1 } }
        # Existing saves used target indexes 1-6 before the shared thought focus.
        if = {
            limit = { check_variable = { SRC_inner_circle_auto_soldiers_target > 0 } }
            add_to_variable = { SRC_inner_circle_auto_soldiers_target = 1 }
        }
        if = {
            limit = { check_variable = { SRC_inner_circle_auto_commissars_target > 0 } }
            add_to_variable = { SRC_inner_circle_auto_commissars_target = 1 }
        }
        if = {
            limit = { check_variable = { SRC_inner_circle_auto_commanders_target > 0 } }
            add_to_variable = { SRC_inner_circle_auto_commanders_target = 1 }
        }
        set_country_flag = SRC_inner_circle_auto_thought_focus_v1
    }
    if = {
        limit = { NOT = { has_country_flag = SRC_inner_circle_project_events_retired } }
        # Clear project choices left by saves from before the events were removed.
        clr_country_flag = SRC_inner_circle_xu_project_a
        clr_country_flag = SRC_inner_circle_xu_project_b
        clr_country_flag = SRC_inner_circle_lan_project_a
        clr_country_flag = SRC_inner_circle_lan_project_b
        clr_country_flag = SRC_inner_circle_guan_project_a
        clr_country_flag = SRC_inner_circle_guan_project_b
        clr_country_flag = SRC_inner_circle_dong_project_a
        clr_country_flag = SRC_inner_circle_dong_project_b
        clr_country_flag = SRC_inner_circle_zhao_project_a
        clr_country_flag = SRC_inner_circle_zhao_project_b
        clr_country_flag = SRC_inner_circle_yang_project_a
        clr_country_flag = SRC_inner_circle_yang_project_b
        clr_country_flag = SRC_inner_circle_xu_fanting_project_a
        clr_country_flag = SRC_inner_circle_xu_fanting_project_b
        clr_country_flag = SRC_inner_circle_he_changgong_project_a
        clr_country_flag = SRC_inner_circle_he_changgong_project_b
        clr_country_flag = SRC_inner_circle_zeng_yiding_project_a
        clr_country_flag = SRC_inner_circle_zeng_yiding_project_b
        clr_country_flag = SRC_inner_circle_liu_yun_project_a
        clr_country_flag = SRC_inner_circle_liu_yun_project_b
        set_country_flag = SRC_inner_circle_project_events_retired

# Reform pauses after logistics until the player chooses an unfinished branch.
SRC_inner_circle_auto_reform_branch_update = {
    if = {
        limit = { check_variable = { SRC_inner_circle_auto_reform_branch = 1 } has_completed_focus = SRC_wanshanshengchantixi }
        set_variable = { SRC_inner_circle_auto_reform_branch = 0 }
    }
    if = {
        limit = { check_variable = { SRC_inner_circle_auto_reform_branch = 2 } has_completed_focus = SRC_shiyanxinshibianzhi }
        set_variable = { SRC_inner_circle_auto_reform_branch = 0 }
    }
    if = {
        limit = { check_variable = { SRC_inner_circle_auto_reform_branch = 3 } has_completed_focus = SRC_qianghuazhuangjiayanjiu }
        set_variable = { SRC_inner_circle_auto_reform_branch = 0 }
    }
    if = {
        limit = {
            has_country_flag = SRC_inner_circle_auto_reform_choice_pending
            OR = {
                NOT = { has_completed_focus = SRC_hongjunhouqinbu }
                NOT = {
                    OR = {
                        has_country_flag = SRC_inner_circle_xu_fanting_active
                        has_country_flag = SRC_inner_circle_he_changgong_active
                    }
                }
                NOT = { check_variable = { SRC_inner_circle_auto_reform_branch = 0 } }
                NOT = {
                    OR = {
                        AND = {
                            NOT = { has_completed_focus = SRC_wanshanshengchantixi }
                            NOT = { has_completed_focus = SRC_shiyanxinshibianzhi }
                        }
                        AND = {
                            NOT = { has_completed_focus = SRC_wanshanshengchantixi }
                            NOT = { has_completed_focus = SRC_qianghuazhuangjiayanjiu }
                        }
                        AND = {
                            NOT = { has_completed_focus = SRC_shiyanxinshibianzhi }
                            NOT = { has_completed_focus = SRC_qianghuazhuangjiayanjiu }
                        }
                    }
                }
            }
        }
        clr_country_flag = SRC_inner_circle_auto_reform_choice_pending
    }
    if = {
        limit = {
            has_completed_focus = SRC_hongjunhouqinbu
            check_variable = { SRC_inner_circle_auto_reform_branch = 0 }
            NOT = { has_country_flag = SRC_inner_circle_auto_reform_choice_pending }
            OR = {
                has_country_flag = SRC_inner_circle_xu_fanting_active
                has_country_flag = SRC_inner_circle_he_changgong_active
            }
        }
        if = {
            limit = {
                OR = {
                    AND = {
                        NOT = { has_completed_focus = SRC_wanshanshengchantixi }
                        NOT = { has_completed_focus = SRC_shiyanxinshibianzhi }
                    }
                    AND = {
                        NOT = { has_completed_focus = SRC_wanshanshengchantixi }
                        NOT = { has_completed_focus = SRC_qianghuazhuangjiayanjiu }
                    }
                    AND = {
                        NOT = { has_completed_focus = SRC_shiyanxinshibianzhi }
                        NOT = { has_completed_focus = SRC_qianghuazhuangjiayanjiu }
                    }
                }
            }
            set_country_flag = SRC_inner_circle_auto_reform_choice_pending
            country_event = { id = SRC_inner_circle_auto.1 days = 1 }
        }
        else_if = {
            limit = {
                NOT = { has_completed_focus = SRC_wanshanshengchantixi }
                has_completed_focus = SRC_shiyanxinshibianzhi
                has_completed_focus = SRC_qianghuazhuangjiayanjiu
            }
            set_variable = { SRC_inner_circle_auto_reform_branch = 1 }
        }
        else_if = {
            limit = {
                NOT = { has_completed_focus = SRC_shiyanxinshibianzhi }
                has_completed_focus = SRC_wanshanshengchantixi
                has_completed_focus = SRC_qianghuazhuangjiayanjiu
            }
            set_variable = { SRC_inner_circle_auto_reform_branch = 2 }
        }
        else_if = {
            limit = {
                NOT = { has_completed_focus = SRC_qianghuazhuangjiayanjiu }
                has_completed_focus = SRC_wanshanshengchantixi
                has_completed_focus = SRC_shiyanxinshibianzhi
            }
            set_variable = { SRC_inner_circle_auto_reform_branch = 3 }
        }
    }
}
