### TOD Party Coalition Effects
### Coalition partner popularity is converted into a stability bonus.
### Formula: partner popularity * ruling-party popularity stability factor * coalition efficiency.
### Usage:
### set_temp_variable = { coalition_partner_var = token:liberal_democracy }
### add_to_coalition = yes
###
### set_temp_variable = { coalition_stability_var = 0.75 }
### set_coalition_stability_liberal_democracy = yes
###
### remove_from_coalition = yes
### end_coalition = yes

TOD_recalculate_coalition_stability_bonus = {
	hidden_effect = {
		TOD_cleanup_invalid_coalition_partners = yes
		TOD_update_coalition_gui_frames = yes
		set_variable = { TOD_coalition_stability_bonus = 0 }
		clear_variable = TOD_coalition_pp_gain
		if = {
			limit = { NOT = { has_variable = TOD_coalition_base_party_popularity_stability_factor } }
			set_variable = { TOD_coalition_base_party_popularity_stability_factor = 0.15 }
		}
		if = {
			limit = { NOT = { has_variable = TOD_coalition_default_efficiency } }
			set_variable = { TOD_coalition_default_efficiency = 0.5 }
		}

		if = {
			limit = { has_country_flag = TOD_coalition_partner_vanguard_socialism }
			set_temp_variable = { TOD_coalition_temp_bonus = party_popularity@vanguard_socialism }
			multiply_temp_variable = { TOD_coalition_temp_bonus = TOD_coalition_base_party_popularity_stability_factor }
			if = {
				limit = { check_variable = { var = TOD_coalition_stability_vanguard_socialism value = 0 compare = greater_than } }
				multiply_temp_variable = { TOD_coalition_temp_bonus = TOD_coalition_stability_vanguard_socialism }
			}
			else = {
				multiply_temp_variable = { TOD_coalition_temp_bonus = TOD_coalition_default_efficiency }
			}
			add_to_variable = { TOD_coalition_stability_bonus = TOD_coalition_temp_bonus }

TOD_add_to_coalition = {
	if = {
		limit = { NOT = { has_variable = coalition_partner_var } }
		log = "ERROR: TOD_add_to_coalition - missing coalition_partner_var"
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { var = coalition_partner_var value = token:vanguard_socialism compare = equals } has_government = vanguard_socialism }
				AND = { check_variable = { var = coalition_partner_var value = token:council_socialism compare = equals } has_government = council_socialism }
				AND = { check_variable = { var = coalition_partner_var value = token:moderate_socialism compare = equals } has_government = moderate_socialism }
				AND = { check_variable = { var = coalition_partner_var value = token:agroindustrial_socialism compare = equals } has_government = agroindustrial_socialism }
				AND = { check_variable = { var = coalition_partner_var value = token:social_democrat compare = equals } has_government = social_democrat }
				AND = { check_variable = { var = coalition_partner_var value = token:liberal_democracy compare = equals } has_government = liberal_democracy }
				AND = { check_variable = { var = coalition_partner_var value = token:consociationalism compare = equals } has_government = consociationalism }
				AND = { check_variable = { var = coalition_partner_var value = token:conservative_democracy compare = equals } has_government = conservative_democracy }
				AND = { check_variable = { var = coalition_partner_var value = token:authoritarian_oligarchy compare = equals } has_government = authoritarian_oligarchy }
				AND = { check_variable = { var = coalition_partner_var value = token:leader_despotism compare = equals } has_government = leader_despotism }
				AND = { check_variable = { var = coalition_partner_var value = token:sacred_revivalism compare = equals } has_government = sacred_revivalism }
				AND = { check_variable = { var = coalition_partner_var value = token:national_corporatism compare = equals } has_government = national_corporatism }
			}
		}
		log = "ERROR: TOD_add_to_coalition - ruling party cannot be a coalition partner"
	}
	else = {
		custom_effect_tooltip = TOD_add_to_coalition_tt
		if = { limit = { check_variable = { var = coalition_partner_var value = token:vanguard_socialism compare = equals } } set_country_flag = TOD_coalition_partner_vanguard_socialism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:council_socialism compare = equals } } set_country_flag = TOD_coalition_partner_council_socialism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:moderate_socialism compare = equals } } set_country_flag = TOD_coalition_partner_moderate_socialism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:agroindustrial_socialism compare = equals } } set_country_flag = TOD_coalition_partner_agroindustrial_socialism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:social_democrat compare = equals } } set_country_flag = TOD_coalition_partner_social_democrat }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:liberal_democracy compare = equals } } set_country_flag = TOD_coalition_partner_liberal_democracy }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:consociationalism compare = equals } } set_country_flag = TOD_coalition_partner_consociationalism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:conservative_democracy compare = equals } } set_country_flag = TOD_coalition_partner_conservative_democracy }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:authoritarian_oligarchy compare = equals } } set_country_flag = TOD_coalition_partner_authoritarian_oligarchy }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:leader_despotism compare = equals } } set_country_flag = TOD_coalition_partner_leader_despotism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:sacred_revivalism compare = equals } } set_country_flag = TOD_coalition_partner_sacred_revivalism }
		if = { limit = { check_variable = { var = coalition_partner_var value = token:national_corporatism compare = equals } } set_country_flag = TOD_coalition_partner_national_corporatism }
		hidden_effect = { TOD_recalculate_coalition_stability_bonus = yes }
	}
}
