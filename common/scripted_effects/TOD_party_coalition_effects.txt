				limit = { has_dynamic_modifier = { modifier = TOD_party_coalition_dynamic_modifier } }

	}

TOD_add_to_coalition = {
	if = {
		limit = { NOT = { has_variable = coalition_partner_var } }
		log = "ERROR: TOD_add_to_coalition - missing coalition_partner_var"
	}
	else_if = {
		limit = {
			OR = {
				AND = { check_variable = { coalition_partner_var = token:vanguard_socialism } has_government = vanguard_socialism }
				AND = { check_variable = { coalition_partner_var = token:council_socialism } has_government = council_socialism }
				AND = { check_variable = { coalition_partner_var = token:moderate_socialism } has_government = moderate_socialism }
				AND = { check_variable = { coalition_partner_var = token:agroindustrial_socialism } has_government = agroindustrial_socialism }
				AND = { check_variable = { coalition_partner_var = token:social_democrat } has_government = social_democrat }
				AND = { check_variable = { coalition_partner_var = token:liberal_democracy } has_government = liberal_democracy }
				AND = { check_variable = { coalition_partner_var = token:consociationalism } has_government = consociationalism }
				AND = { check_variable = { coalition_partner_var = token:conservative_democracy } has_government = conservative_democracy }
				AND = { check_variable = { coalition_partner_var = token:authoritarian_oligarchy } has_government = authoritarian_oligarchy }
				AND = { check_variable = { coalition_partner_var = token:leader_despotism } has_government = leader_despotism }
				AND = { check_variable = { coalition_partner_var = token:sacred_revivalism } has_government = sacred_revivalism }
				AND = { check_variable = { coalition_partner_var = token:national_corporatism } has_government = national_corporatism }
			}
		}
		log = "ERROR: TOD_add_to_coalition - ruling party cannot be a coalition partner"
	}
	else = {
		custom_effect_tooltip = TOD_add_to_coalition_tt
		if = { limit = { check_variable = { coalition_partner_var = token:vanguard_socialism } } set_country_flag = TOD_coalition_partner_vanguard_socialism }
		if = { limit = { check_variable = { coalition_partner_var = token:council_socialism } } set_country_flag = TOD_coalition_partner_council_socialism }
		if = { limit = { check_variable = { coalition_partner_var = token:moderate_socialism } } set_country_flag = TOD_coalition_partner_moderate_socialism }
		if = { limit = { check_variable = { coalition_partner_var = token:agroindustrial_socialism } } set_country_flag = TOD_coalition_partner_agroindustrial_socialism }
		if = { limit = { check_variable = { coalition_partner_var = token:social_democrat } } set_country_flag = TOD_coalition_partner_social_democrat }
		if = { limit = { check_variable = { coalition_partner_var = token:liberal_democracy } } set_country_flag = TOD_coalition_partner_liberal_democracy }
		if = { limit = { check_variable = { coalition_partner_var = token:consociationalism } } set_country_flag = TOD_coalition_partner_consociationalism }
		if = { limit = { check_variable = { coalition_partner_var = token:conservative_democracy } } set_country_flag = TOD_coalition_partner_conservative_democracy }
		if = { limit = { check_variable = { coalition_partner_var = token:authoritarian_oligarchy } } set_country_flag = TOD_coalition_partner_authoritarian_oligarchy }
		if = { limit = { check_variable = { coalition_partner_var = token:leader_despotism } } set_country_flag = TOD_coalition_partner_leader_despotism }
		if = { limit = { check_variable = { coalition_partner_var = token:sacred_revivalism } } set_country_flag = TOD_coalition_partner_sacred_revivalism }
		if = { limit = { check_variable = { coalition_partner_var = token:national_corporatism } } set_country_flag = TOD_coalition_partner_national_corporatism }
		hidden_effect = { TOD_recalculate_coalition_pp_gain = yes }
	}
}

TOD_end_coalition = {
	custom_effect_tooltip = TOD_end_coalition_tt
	hidden_effect = {
		clr_country_flag = TOD_coalition_partner_vanguard_socialism
		clr_country_flag = TOD_coalition_partner_council_socialism
		clr_country_flag = TOD_coalition_partner_moderate_socialism
		clr_country_flag = TOD_coalition_partner_agroindustrial_socialism
		clr_country_flag = TOD_coalition_partner_social_democrat
		clr_country_flag = TOD_coalition_partner_liberal_democracy
		clr_country_flag = TOD_coalition_partner_consociationalism
		clr_country_flag = TOD_coalition_partner_conservative_democracy
		clr_country_flag = TOD_coalition_partner_authoritarian_oligarchy
		clr_country_flag = TOD_coalition_partner_leader_despotism
		clr_country_flag = TOD_coalition_partner_sacred_revivalism
		clr_country_flag = TOD_coalition_partner_national_corporatism
		clear_variable = TOD_coalition_pp_gain
		clear_variable = TOD_coalition_stability_vanguard_socialism
		clear_variable = TOD_coalition_stability_council_socialism
		clear_variable = TOD_coalition_stability_moderate_socialism
		clear_variable = TOD_coalition_stability_agroindustrial_socialism
		clear_variable = TOD_coalition_stability_social_democrat
		clear_variable = TOD_coalition_stability_liberal_democracy
		clear_variable = TOD_coalition_stability_consociationalism
		clear_variable = TOD_coalition_stability_conservative_democracy
		clear_variable = TOD_coalition_stability_authoritarian_oligarchy
		clear_variable = TOD_coalition_stability_leader_despotism
		clear_variable = TOD_coalition_stability_sacred_revivalism
		clear_variable = TOD_coalition_stability_national_corporatism
		if = {
			limit = { has_dynamic_modifier = { modifier = TOD_party_coalition_dynamic_modifier } }
			remove_dynamic_modifier = { modifier = TOD_party_coalition_dynamic_modifier }
		}
		force_update_dynamic_modifier = yes
	}
}
