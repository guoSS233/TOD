# Settle an already-expired clock before a lifecycle operation can reset its owner or performance.
# A one-day rearm barrier prevents a late callback for this native ID from touching its successor.
ROC_ma_settle_due = {
    # Preserve original three-stage durations until the six-stage conversion is complete.
    if = {
        limit = { NOT = { has_country_flag = ROC_ma_five_stage_migrated } NOT = { has_country_flag = ROC_ma_six_stage_migrated } }
        set_country_flag = ROC_ma_three_stage_conversion_pending
    }
    ROC_ma_migrate_five_stages = yes
    ROC_ma_migrate_six_stages = yes
    clr_country_flag = ROC_ma_three_stage_conversion_pending
    if = {
        limit = {
            check_variable = { ROC_ma_clock_owner > 0 }
            NOT = { has_country_flag = ROC_ma_rearm_blocked }
        }
        if = {
            limit = { has_active_mission = ROC_ma_current_action }
            set_variable = { ROC_ma_days_remaining = days_mission_timeout@ROC_ma_current_action }
            set_country_flag = ROC_ma_clock_armed
        }
        if = {
            limit = {
                OR = {
                    has_country_flag = ROC_ma_timeout_pending
                    AND = {
                        has_active_mission = ROC_ma_current_action
                        check_variable = { var = ROC_ma_days_remaining value = 0 compare = less_than_or_equals }
                    }
                    AND = {
                        NOT = { has_active_mission = ROC_ma_current_action }
                        has_country_flag = ROC_ma_clock_armed
                        check_variable = { var = ROC_ma_days_remaining value = 1 compare = less_than_or_equals }
                    }
                }
            }
            set_country_flag = { flag = ROC_ma_rearm_blocked days = 1 }
            clr_country_flag = ROC_ma_timeout_pending
            clr_country_flag = ROC_ma_clock_armed
            remove_mission = ROC_ma_current_action
            ROC_ma_finish_action = yes
        }
    }
}
