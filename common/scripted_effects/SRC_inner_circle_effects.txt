# The portrait buttons guard eligibility before setting pending. Recheck the shared
# appointment conditions here so a stale UI click cannot charge or replace a seat.
SRC_inner_circle_assign_selected = {
    if = {
        limit = {
            SRC_inner_circle_slot_selected = yes
            SRC_inner_circle_appointment_available = yes
            check_variable = { SRC_inner_circle_pending > 0 }
        }
        SRC_inner_circle_initialize = yes
        SRC_inner_circle_migrate_layout_v2 = yes
        add_political_power = -50
        if = {
            limit = { check_variable = { SRC_inner_circle_selected_slot = 1 } }
            SRC_inner_circle_remove_slot_1 = yes
            set_variable = { SRC_inner_circle_slot_1 = SRC_inner_circle_pending }
        }
        else_if = {
            limit = { check_variable = { SRC_inner_circle_selected_slot = 2 } }
            SRC_inner_circle_remove_slot_2 = yes
            set_variable = { SRC_inner_circle_slot_2 = SRC_inner_circle_pending }
        }
        else_if = {
            limit = { check_variable = { SRC_inner_circle_selected_slot = 3 } }
            SRC_inner_circle_remove_slot_3 = yes
            set_variable = { SRC_inner_circle_slot_3 = SRC_inner_circle_pending }
        }
        else = {
            SRC_inner_circle_remove_slot_4 = yes
            set_variable = { SRC_inner_circle_slot_4 = SRC_inner_circle_pending }
        }
        SRC_inner_circle_add_pending = yes
        set_variable = { SRC_inner_circle_selected_slot = 0 }
    }
    set_variable = { SRC_inner_circle_pending = 0 }
}

SRC_inner_circle_migrate_layout_v2 = {
    if = {
        limit = { NOT = { has_country_flag = SRC_inner_circle_layout_v2 } }
        if = {
            limit = { check_variable = { SRC_inner_circle_slot_3 > 0 } }
            if = {
                limit = { check_variable = { SRC_inner_circle_slot_1 = 0 } }
                set_variable = { SRC_inner_circle_slot_1 = SRC_inner_circle_slot_3 }
            }
            else_if = {
                limit = { check_variable = { SRC_inner_circle_slot_2 = 0 } }
                set_variable = { SRC_inner_circle_slot_2 = SRC_inner_circle_slot_3 }
            }
            else = {
                SRC_inner_circle_remove_legacy_slot_3 = yes
                add_political_power = 50
            }
        }
        set_variable = { SRC_inner_circle_slot_3 = 0 }
        set_variable = { SRC_inner_circle_slot_4 = 0 }
        set_country_flag = SRC_inner_circle_layout_v2
    }
}
