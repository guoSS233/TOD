CZE_show_current_skoda_prioty_idea_stats_infantry_focus = {
	if = {
		limit = {
			NOT = {
				OR = {
					has_idea = CZE_skoda_infantry_focus_optimization_1_idea
					has_idea = CZE_skoda_infantry_focus_optimization_2_idea
					has_idea = CZE_skoda_infantry_focus_optimization_3_idea
					has_idea = CZE_skoda_infantry_focus_optimization_4_idea
					has_idea = CZE_skoda_infantry_focus_optimization_5_idea
					has_idea = CZE_skoda_infantry_focus_optimization_6_idea
					has_idea = CZE_skoda_infantry_focus_optimization_7_idea
				}
			}
		}
		custom_effect_tooltip = CZE_skoda_set_sptimization_tier_to_one_tt
		effect_tooltip = {
			add_ideas = CZE_skoda_infantry_focus_optimization_1_idea
		}
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_infantry_focus_optimization_1_idea
		}
		show_ideas_tooltip = CZE_skoda_no_focus_optimization_1_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_infantry_focus_optimization_2_idea
		}
		show_ideas_tooltip = CZE_skoda_infantry_focus_optimization_2_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_infantry_focus_optimization_3_idea
		}
		show_ideas_tooltip = CZE_skoda_infantry_focus_optimization_3_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_infantry_focus_optimization_4_idea
		}
		show_ideas_tooltip = CZE_skoda_infantry_focus_optimization_4_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_infantry_focus_optimization_5_idea
		}
		show_ideas_tooltip = CZE_skoda_infantry_focus_optimization_5_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_no_focus_optimization_6_idea
		}
		show_ideas_tooltip = CZE_skoda_no_focus_optimization_6_idea
	}
	else_if = {
		limit = {
			has_idea = CZE_skoda_no_focus_optimization_7_idea
		}
		show_ideas_tooltip = CZE_skoda_no_focus_optimization_7_idea
	}
	else = { #if something broke I guess we're on 1?
		show_ideas_tooltip = CZE_skoda_no_focus_optimization_1_idea
	}
}
