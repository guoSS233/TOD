ROC_bill_refresh_political_support = {
    if = {
        limit = { OR = { check_variable = { ROC_bill_stage = 1 } check_variable = { ROC_bill_stage = 2 } } }
        # Legacy variable names remain stable for old saves and localisation.
        set_variable = { ROC_bill_political_support = 0 }
        set_variable = { ROC_bill_requirement = 2 }
        if = { limit = { OR = { check_variable = { ROC_bill_id = 2 } check_variable = { ROC_bill_id = 4 } check_variable = { ROC_bill_id = 6 } check_variable = { ROC_bill_id = 8 } check_variable = { ROC_bill_id = 10 } } } set_variable = { ROC_bill_requirement = 3 } }
        if = { limit = { check_variable = { ROC_bill_tactic = 1 } } add_to_variable = { var = ROC_bill_political_support value = 2 } }
        else_if = { limit = { check_variable = { ROC_bill_tactic = 2 } } add_to_variable = { var = ROC_bill_political_support value = 1 } }
        if = { limit = { ROC_bill_economic_policy_basis = yes } add_to_variable = { var = ROC_bill_political_support value = 1 } }
        if = { limit = { check_variable = { var = ROC_econ_stage value = 2 compare = greater_than_or_equals } } add_to_variable = { var = ROC_bill_political_support value = 1 } }
        if = { limit = { has_country_flag = ROC_bill_legacy_administration } clamp_variable = { var = ROC_bill_political_support min = 2 max = 4 } }
        if = { limit = { has_country_flag = ROC_bill_supervision } add_to_variable = { var = ROC_bill_requirement value = 1 } }
        set_variable = { ROC_bill_conditional_requirement = ROC_bill_requirement }
        subtract_from_variable = { var = ROC_bill_conditional_requirement value = 1 }
        set_variable = { ROC_bill_support = ROC_bill_political_support }
    }
}
