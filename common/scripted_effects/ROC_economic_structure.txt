# Ordinary economic structure. Never activates a native balance or political race.
ROC_ce_policy_capture_legacy_pending = {
    # Legacy effects wrote fractions; the ordinary policy queue uses percentage points.
    set_variable = { ROC_ce_policy_legacy_pending = ROC_econ_pending_shift }
    multiply_variable = { var = ROC_ce_policy_legacy_pending value = 100 }
    add_to_variable = { var = ROC_ce_policy_pending value = ROC_ce_policy_legacy_pending }
    set_variable = { ROC_econ_pending_shift = 0 }
    clamp_variable = { var = ROC_ce_policy_pending min = -30 max = 30 }
}

ROC_ce_policy_initialize = {
    if = {
        limit = { country_exists = ROC original_tag = ROC }
        if = {
            limit = { NOT = { has_country_flag = ROC_ce_policy_migrated } }
            set_variable = { ROC_ce_policy_axis = 0 }
            if = {
                limit = { has_power_balance = { id = ROC_power_balance } }
                # Capture before ROC_ce_retire_bop removes the saved native value.
                set_variable = { ROC_ce_policy_axis = power_balance_value }
                multiply_variable = { var = ROC_ce_policy_axis value = 100 }
            }
            set_country_flag = ROC_ce_policy_migrated
        }
        # Old pending points have mixed provenance and are deliberately kept intact.
        if = {
            limit = { NOT = { has_country_flag = ROC_ma_policy_ledger_initialized } }
            set_country_flag = ROC_ma_policy_ledger_initialized
            set_variable = { ROC_ma_policy_pending = 0 }
            set_variable = { ROC_ma_policy_owner = 0 }
        }
        ROC_ce_policy_capture_legacy_pending = yes
        ROC_ce_policy_refresh = yes
    }
}

ROC_ce_policy_remove_ideas = {
    remove_ideas = ROC_ce_structure_mixed
    remove_ideas = ROC_ce_structure_market_1
    remove_ideas = ROC_ce_structure_market_2
    remove_ideas = ROC_ce_structure_market_3
    remove_ideas = ROC_ce_structure_public_1
    remove_ideas = ROC_ce_structure_public_2
    remove_ideas = ROC_ce_structure_public_3
}

ROC_ce_policy_refresh = {
    if = {
        limit = { country_exists = ROC ROC_ce_administration_active = yes }
        clamp_variable = { var = ROC_ce_policy_axis min = -100 max = 100 }
        set_variable = { ROC_ce_policy_frame = ROC_ce_policy_axis }
        add_to_variable = { var = ROC_ce_policy_frame value = 100 }
        divide_variable = { var = ROC_ce_policy_frame value = 2 }
        clamp_variable = { var = ROC_ce_policy_pending min = -30 max = 30 }
        clamp_variable = { var = ROC_ma_policy_pending min = -30 max = 30 }
        set_variable = { ROC_ce_policy_tier = 0 }
        if = { limit = { check_variable = { ROC_ce_policy_axis > 75 } } set_variable = { ROC_ce_policy_tier = 3 } }
        else_if = { limit = { check_variable = { ROC_ce_policy_axis > 50 } } set_variable = { ROC_ce_policy_tier = 2 } }
        else_if = { limit = { check_variable = { ROC_ce_policy_axis > 25 } } set_variable = { ROC_ce_policy_tier = 1 } }
        else_if = { limit = { check_variable = { ROC_ce_policy_axis < -75 } } set_variable = { ROC_ce_policy_tier = -3 } }
        else_if = { limit = { check_variable = { ROC_ce_policy_axis < -50 } } set_variable = { ROC_ce_policy_tier = -2 } }
        else_if = { limit = { check_variable = { ROC_ce_policy_axis < -25 } } set_variable = { ROC_ce_policy_tier = -1 } }
        set_variable = { ROC_ce_policy_idea_count = 0 }
        if = { limit = { has_idea = ROC_ce_structure_mixed } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_market_1 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_market_2 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_market_3 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_public_1 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_public_2 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = { limit = { has_idea = ROC_ce_structure_public_3 } add_to_variable = { var = ROC_ce_policy_idea_count value = 1 } }
        if = {
            limit = {
                OR = {
                    NOT = { has_country_flag = ROC_ce_policy_effects_active }
                    NOT = { check_variable = { var = ROC_ce_policy_applied_tier value = ROC_ce_policy_tier compare = equals } }
                    NOT = { check_variable = { var = ROC_ce_policy_idea_count value = 1 compare = equals } }
                    AND = { check_variable = { ROC_ce_policy_tier = 0 } NOT = { has_idea = ROC_ce_structure_mixed } }
                    AND = { check_variable = { ROC_ce_policy_tier = -1 } NOT = { has_idea = ROC_ce_structure_market_1 } }
                    AND = { check_variable = { ROC_ce_policy_tier = -2 } NOT = { has_idea = ROC_ce_structure_market_2 } }
                    AND = { check_variable = { ROC_ce_policy_tier = -3 } NOT = { has_idea = ROC_ce_structure_market_3 } }
                    AND = { check_variable = { ROC_ce_policy_tier = 1 } NOT = { has_idea = ROC_ce_structure_public_1 } }
                    AND = { check_variable = { ROC_ce_policy_tier = 2 } NOT = { has_idea = ROC_ce_structure_public_2 } }
                    AND = { check_variable = { ROC_ce_policy_tier = 3 } NOT = { has_idea = ROC_ce_structure_public_3 } }
                }
            }
            ROC_ce_policy_remove_ideas = yes
            if = { limit = { check_variable = { ROC_ce_policy_tier = 0 } } add_ideas = ROC_ce_structure_mixed }
            if = { limit = { check_variable = { ROC_ce_policy_tier = -1 } } add_ideas = ROC_ce_structure_market_1 }
            if = { limit = { check_variable = { ROC_ce_policy_tier = -2 } } add_ideas = ROC_ce_structure_market_2 }
            if = { limit = { check_variable = { ROC_ce_policy_tier = -3 } } add_ideas = ROC_ce_structure_market_3 }
            if = { limit = { check_variable = { ROC_ce_policy_tier = 1 } } add_ideas = ROC_ce_structure_public_1 }
            if = { limit = { check_variable = { ROC_ce_policy_tier = 2 } } add_ideas = ROC_ce_structure_public_2 }
            if = { limit = { check_variable = { ROC_ce_policy_tier = 3 } } add_ideas = ROC_ce_structure_public_3 }
            set_variable = { ROC_ce_policy_applied_tier = ROC_ce_policy_tier }
            set_country_flag = ROC_ce_policy_effects_active
        }
        # These multipliers affect legacy state construction, never central project speed.
        set_variable = { VAR_ROC_jianshe_days_factor = 1 }
        set_variable = { VAR_ROC_jianshe_pp_factor = 1 }
        if = { limit = { check_variable = { ROC_ce_policy_tier = -3 } } set_variable = { VAR_ROC_jianshe_days_factor = 1.5 } }
        if = { limit = { check_variable = { ROC_ce_policy_tier = 3 } } set_variable = { VAR_ROC_jianshe_pp_factor = 1.5 } }
        ROC_scripted_effects_jianshe = yes
    }
    else = {
        ROC_ce_policy_remove_ideas = yes
        clr_country_flag = ROC_ce_policy_effects_active
    }
}

ROC_ce_policy_apply_monthly = {
    ROC_ce_policy_initialize = yes
    if = {
        limit = { country_exists = ROC ROC_ce_administration_active = yes }
        set_variable = { ROC_ce_policy_step = ROC_ce_policy_pending }
        if = {
            limit = { NOT = { check_variable = { ROC_ma_policy_pending = 0 } } }
            set_variable = { ROC_ce_policy_step = ROC_ma_policy_pending }
        }
        clamp_variable = { var = ROC_ce_policy_step min = -1 max = 1 }
        set_variable = { ROC_ce_policy_lower_room = -100 }
        subtract_from_variable = { var = ROC_ce_policy_lower_room value = ROC_ce_policy_axis }
        set_variable = { ROC_ce_policy_upper_room = 100 }
        subtract_from_variable = { var = ROC_ce_policy_upper_room value = ROC_ce_policy_axis }
        clamp_variable = { var = ROC_ce_policy_step min = ROC_ce_policy_lower_room max = ROC_ce_policy_upper_room }
        add_to_variable = { var = ROC_ce_policy_axis value = ROC_ce_policy_step }
        if = {
            limit = { NOT = { check_variable = { ROC_ma_policy_pending = 0 } } }
            subtract_from_variable = { var = ROC_ma_policy_pending value = ROC_ce_policy_step }
        }
        else = { subtract_from_variable = { var = ROC_ce_policy_pending value = ROC_ce_policy_step } }
        if = { limit = { check_variable = { var = ROC_ce_policy_axis value = 100 compare = greater_than_or_equals } check_variable = { ROC_ce_policy_pending > 0 } } set_variable = { ROC_ce_policy_pending = 0 } }
        if = { limit = { check_variable = { var = ROC_ce_policy_axis value = -100 compare = less_than_or_equals } check_variable = { ROC_ce_policy_pending < 0 } } set_variable = { ROC_ce_policy_pending = 0 } }
        if = { limit = { check_variable = { var = ROC_ce_policy_axis value = 100 compare = greater_than_or_equals } check_variable = { ROC_ma_policy_pending > 0 } } set_variable = { ROC_ma_policy_pending = 0 } }
        if = { limit = { check_variable = { var = ROC_ce_policy_axis value = -100 compare = less_than_or_equals } check_variable = { ROC_ma_policy_pending < 0 } } set_variable = { ROC_ma_policy_pending = 0 } }
        ROC_ce_policy_refresh = yes
    }
}
