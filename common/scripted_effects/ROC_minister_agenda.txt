# Convert five-stage clocks once; unfinished stages keep their completed proportion.
ROC_ma_migrate_six_stages = {
    if = {
        limit = { NOT = { has_country_flag = ROC_ma_six_stage_migrated } }
        set_country_flag = ROC_ma_six_stage_migrated
        if = {
            limit = { has_active_mission = ROC_ma_current_action }
            set_variable = { ROC_ma_days_remaining = days_mission_timeout@ROC_ma_current_action }
        }
        # Capture an already-due old focus before rescaling a saved last day.
        if = {
            limit = {
                check_variable = { ROC_ma_clock_owner > 0 }
                check_variable = { ROC_ma_stage > 0 }
                check_variable = { ROC_ma_stage < 6 }
                OR = {
                    has_country_flag = ROC_ma_timeout_pending
                    AND = { has_active_mission = ROC_ma_current_action check_variable = { var = ROC_ma_days_remaining value = 0 compare = less_than_or_equals } }
                    AND = { NOT = { has_active_mission = ROC_ma_current_action } has_country_flag = ROC_ma_clock_armed check_variable = { var = ROC_ma_days_remaining value = 1 compare = less_than_or_equals } }
                }
            }
            set_country_flag = ROC_ma_timeout_pending
        }
        # Settle a due old routine once before stage 6 becomes the new focus.
        if = {
            limit = { check_variable = { ROC_ma_clock_owner > 0 } check_variable = { ROC_ma_stage = 6 } }
            if = {
                limit = {
                    # An unarmed callback in the rearm barrier belongs to the old task.
                    OR = { NOT = { has_country_flag = ROC_ma_rearm_blocked } has_country_flag = ROC_ma_clock_armed }
                    OR = {
                        AND = {
                            has_active_mission = ROC_ma_current_action
                            check_variable = { var = ROC_ma_days_remaining value = 0 compare = less_than_or_equals }
                        }
                        AND = {
                            NOT = { has_active_mission = ROC_ma_current_action }
                            check_variable = { var = ROC_ma_days_remaining value = 1 compare = less_than_or_equals }
                            OR = { has_country_flag = ROC_ma_timeout_pending has_country_flag = ROC_ma_clock_armed }
                        }
                    }
                }
                add_to_variable = { var = ROC_ce_actual value = 10 }
            }
            clr_country_flag = ROC_ma_timeout_pending
        }
        else_if = {
            # An orphan timeout has no minister to credit and must not block rearming.
            limit = { check_variable = { var = ROC_ma_clock_owner value = 0 compare = less_than_or_equals } }
            clr_country_flag = ROC_ma_timeout_pending
        }
        ROC_ma_save_owner = yes
        if = { limit = { check_variable = { ROC_ma_yu_stage = 6 } } set_variable = { ROC_ma_yu_days = 180 } }
        else_if = {
            limit = { check_variable = { ROC_ma_yu_stage > 0 } check_variable = { ROC_ma_yu_stage < 6 } check_variable = { ROC_ma_yu_days > 0 } }
            multiply_variable = { var = ROC_ma_yu_days value = 180 }
            if = {
                limit = { has_country_flag = ROC_ma_three_stage_conversion_pending }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 1 } } divide_variable = { var = ROC_ma_yu_days value = 210 } }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 3 } } divide_variable = { var = ROC_ma_yu_days value = 280 } }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 5 } } divide_variable = { var = ROC_ma_yu_days value = 350 } }
            }
            else = {
                if = { limit = { check_variable = { ROC_ma_yu_stage = 1 } } divide_variable = { var = ROC_ma_yu_days value = 70 } }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 2 } } divide_variable = { var = ROC_ma_yu_days value = 84 } }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 3 } } divide_variable = { var = ROC_ma_yu_days value = 98 } }
                if = { limit = { check_variable = { ROC_ma_yu_stage = 4 } } divide_variable = { var = ROC_ma_yu_days value = 112 } }

            round_variable = ROC_ma_yu_days
            clamp_variable = { var = ROC_ma_yu_days min = 1 max = 99999 }
        }
        if = { limit = { check_variable = { ROC_ma_yan_stage = 6 } } set_variable = { ROC_ma_yan_days = 180 } }
        else_if = {
            limit = { check_variable = { ROC_ma_yan_stage > 0 } check_variable = { ROC_ma_yan_stage < 6 } check_variable = { ROC_ma_yan_days > 0 } }
            multiply_variable = { var = ROC_ma_yan_days value = 180 }
            if = {
                limit = { has_country_flag = ROC_ma_three_stage_conversion_pending }
                if = { limit = { check_variable = { ROC_ma_yan_stage = 1 } } divide_variable = { var = ROC_ma_yan_days value = 210 } }

# Settle an already-expired clock before a lifecycle operation can reset its owner or performance.
# A one-day rearm barrier prevents a late callback for this native ID from touching its successor.
ROC_ma_settle_due = {
    # Preserve original three-stage durations until the six-stage conversion is complete.
    if = {
        limit = { NOT = { has_country_flag = ROC_ma_five_stage_migrated } NOT = { has_country_flag = ROC_ma_six_stage_migrated } }
        set_country_flag = ROC_ma_three_stage_conversion_pending
    }
    ROC_ma_migrate_five_stages = yes
    ROC_ma_migrate_six_stages = yes
    clr_country_flag = ROC_ma_three_stage_conversion_pending
    if = {
        limit = {
            check_variable = { ROC_ma_clock_owner > 0 }
            NOT = { has_country_flag = ROC_ma_rearm_blocked }
        }
        if = {
            limit = { has_active_mission = ROC_ma_current_action }
            set_variable = { ROC_ma_days_remaining = days_mission_timeout@ROC_ma_current_action }
            set_country_flag = ROC_ma_clock_armed
        }
        if = {
            limit = {
                OR = {
                    has_country_flag = ROC_ma_timeout_pending
                    AND = {
                        has_active_mission = ROC_ma_current_action
                        check_variable = { var = ROC_ma_days_remaining value = 0 compare = less_than_or_equals }
                    }
                    AND = {
                        NOT = { has_active_mission = ROC_ma_current_action }
                        has_country_flag = ROC_ma_clock_armed
                        check_variable = { var = ROC_ma_days_remaining value = 1 compare = less_than_or_equals }
                    }
                }
            }
            set_country_flag = { flag = ROC_ma_rearm_blocked days = 1 }
            clr_country_flag = ROC_ma_timeout_pending
            clr_country_flag = ROC_ma_clock_armed
            remove_mission = ROC_ma_current_action
            ROC_ma_finish_action = yes
        }
    }
}

# Exact excess credit crosses stages and subsequent 90-day execution cycles.
ROC_ma_advance_days = {
    ROC_ma_sync = yes
    if = {
        limit = {
            ROC_ma_can_run = yes
            NOT = { has_country_flag = ROC_ma_timeout_pending }
            check_variable = { ROC_ma_day_credit > 0 }
        }
        remove_mission = ROC_ma_current_action
        clr_country_flag = ROC_ma_clock_armed
        while_loop_effect = {
            limit = {
                check_variable = { var = ROC_ma_day_credit value = ROC_ma_days_remaining compare = greater_than_or_equals }
            }
            subtract_from_variable = { var = ROC_ma_day_credit value = ROC_ma_days_remaining }
            ROC_ma_finish_action = yes
        }
        subtract_from_variable = { var = ROC_ma_days_remaining value = ROC_ma_day_credit }
        set_variable = { ROC_ma_day_credit = 0 }
        ROC_ma_save_owner = yes
        ROC_ma_sync = yes
    }
    set_variable = { ROC_ma_day_credit = 0 }
}

ROC_ma_delay_days = {
    ROC_ma_sync = yes
    if = {
        limit = {
            ROC_ma_can_run = yes
            NOT = { has_country_flag = ROC_ma_timeout_pending }
            check_variable = { ROC_ma_day_delay > 0 }
        }
        add_to_variable = { var = ROC_ma_days_remaining value = ROC_ma_day_delay }
        if = {
            limit = { has_active_mission = ROC_ma_current_action }
            add_days_mission_timeout = { mission = ROC_ma_current_action days = ROC_ma_day_delay }
        }
        ROC_ma_save_owner = yes
    }
    set_variable = { ROC_ma_day_delay = 0 }
}
