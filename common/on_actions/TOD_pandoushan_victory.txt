# ROOT is the capitulated country; FROM is the country credited with the victory.
# Each defeated country is settled independently so the remaining front can continue.
on_actions = {
on_capitulation = {
	effect = {
		# 拉普拉塔危机：阿根廷投降后按协约安排转移边境州并结束与阿根廷的战争。
		if = {
			limit = {
				ROOT = {
					tag = ARG
					has_country_flag = TOD_ARG_la_plata_crisis_started
				}
				has_global_flag = TOD_nanmei_pandoushan_pact_signed
				FROM = { OR = { tag = CHL tag = PAR tag = UPB is_in_faction_with = CHL } }
				CHL = { NOT = { has_country_flag = TOD_CHL_la_plata_victory_settlement_applied } }
			}
			CHL = {
				set_country_flag = TOD_CHL_la_plata_victory_settlement_applied
				if = {
					limit = { country_exists = PAR }
					510 = { add_core_of = PAR }
					PAR = { transfer_state = 510 }
				}
				else = {
					510 = { add_core_of = CHL }
					transfer_state = 510
				}
				511 = { add_core_of = CHL }
				transfer_state = 511
				ROOT = { set_cosmetic_tag = ARG_la_plata_transitional_government }
				if = { limit = { has_war_with = ARG } white_peace = ARG }
			}
			PAR = { if = { limit = { has_war_with = ARG } white_peace = ARG } }
			UPB = { if = { limit = { has_war_with = ARG } white_peace = ARG } }
		}

		# 巴西投降：圣保罗交予巴拉圭，南里奥格兰德与巴伊亚分别独立。
		if = {
			limit = {
				ROOT = { tag = BRA }
				has_global_flag = TOD_nanmei_pandoushan_revenge_war_started
				FROM = { OR = { tag = CHL tag = PAR tag = UPB is_in_faction_with = CHL } }
				CHL = { NOT = { has_country_flag = TOD_CHL_brazil_victory_settlement_applied } }
			}
			CHL = {
				set_country_flag = TOD_CHL_brazil_victory_settlement_applied
				if = {
					limit = { country_exists = PAR }
					501 = { add_core_of = PAR }
					PAR = { transfer_state = 501 }
				}
				else = {
					501 = { add_core_of = CHL }
					transfer_state = 501
				}
				498 = { add_core_of = RGR }
				499 = { add_core_of = BAI }
				transfer_state = 498
				transfer_state = 499
				if = {
					limit = { country_exists = RGR }
					RGR = { transfer_state = 498 }
				}
				else = { release = RGR }
				if = {
					limit = { country_exists = BAI }
					BAI = { transfer_state = 499 }
				}
				else = { release = BAI }
			}
		}

		# 哥伦比亚投降：按文档处理巴拿马、四方王国边境州，并将东哥伦比亚恢复为独立委内瑞拉。
		if = {
			limit = {
				ROOT = { tag = COL }
				has_global_flag = TOD_nanmei_pandoushan_revenge_war_started
				FROM = { OR = { tag = CHL tag = PAR tag = UPB is_in_faction_with = CHL } }
				CHL = { NOT = { has_country_flag = TOD_CHL_colombia_victory_settlement_applied } }
			}
			CHL = {
				set_country_flag = TOD_CHL_colombia_victory_settlement_applied
				304 = { add_core_of = CHL }
				transfer_state = 304
				if = {
					limit = { country_exists = UPB }
					486 = { add_core_of = UPB }
					UPB = { transfer_state = 486 }
				}
				else = {
					486 = { add_core_of = CHL }
					transfer_state = 486
				}
				307 = { add_core_of = VEN }
				488 = { add_core_of = VEN }
				489 = { add_core_of = VEN }
				if = {
					limit = { country_exists = VEN }
					VEN = {
						transfer_state = 307
						transfer_state = 488
						transfer_state = 489
						if = {
							limit = { is_subject_of = COL }
							COL = { end_puppet = VEN }
						}
						if = { limit = { is_in_faction = yes } leave_faction = yes }
						set_cosmetic_tag = VEN_venezuelan_republic
					}
				}
				else = {
					transfer_state = 307
					transfer_state = 488
					transfer_state = 489
					release = VEN
					VEN = { set_cosmetic_tag = VEN_venezuelan_republic }
				}
			}
		}

		# 巴西与哥伦比亚都完成各自的投降结算后，才结束整场复仇战争。
		if = {
			limit = {
				has_global_flag = TOD_nanmei_pandoushan_revenge_war_started
				ROOT = { OR = { tag = BRA tag = COL } }
				FROM = { OR = { tag = CHL tag = PAR tag = UPB is_in_faction_with = CHL } }
				CHL = {
					OR = { has_country_flag = TOD_CHL_brazil_victory_settlement_applied NOT = { country_exists = BRA } }
					OR = { has_country_flag = TOD_CHL_colombia_victory_settlement_applied NOT = { country_exists = COL } }
					NOT = { has_country_flag = TOD_CHL_pandoushan_revenge_war_closed }
				}
			}
			CHL = {
				set_country_flag = TOD_CHL_pandoushan_revenge_war_closed
				if = { limit = { has_war_with = BRA } white_peace = BRA }
				if = { limit = { has_war_with = COL } white_peace = COL }
				if = { limit = { has_war_with = VEN } white_peace = VEN }
			}
			PAR = {
				if = { limit = { has_war_with = BRA } white_peace = BRA }
				if = { limit = { has_war_with = COL } white_peace = COL }
				if = { limit = { has_war_with = VEN } white_peace = VEN }
			}
			UPB = {
				if = { limit = { has_war_with = BRA } white_peace = BRA }
				if = { limit = { has_war_with = COL } white_peace = COL }
				if = { limit = { has_war_with = VEN } white_peace = VEN }
			}
		}
	}
}
}
